function doPost(e) {
  let body;
  try { body = JSON.parse(e.postData.contents); } catch (err) { return json({ error: "bad json" }); }
  if (FAMILY_CODE === "CHANGE-ME" || typeof body.key !== "string" || body.key !== FAMILY_CODE) {
    Utilities.sleep(1500);                       // slows down anyone guessing codes
    return json({ error: "bad key" });
  }
  if (body.action === "events") return json({ events: getEvents() });
  if (body.action === "state") {
    const days = Math.min(Number(body.days) || 8, 31);
    return json({ checks: readChecks(body.date), logs: readLogs(body.date, days), shop: readShop(),
                  shifts: readShifts(), work: readWork(body.date), countdowns: readCountdowns() });
  }
  if (body.op && String(body.op).indexOf("shop") === 0) return handleShopOp(body);
  if (body.op && String(body.op).indexOf("shift") === 0) return handleShiftOp(body);
  if (body.op && String(body.op).indexOf("cd") === 0) return handleCountdownOp(body);
  if (body.op) return handleLogOp(body);

  // iPhone Shortcuts / automations: {"key", "action":"check", "k": kid, "t": task}
  if (body.action === "check") return handleCheck(body);
  if (body.action === "work") return handleWork(body);
  if (body.action === "recipe") return handleRecipe(body);

  // Dashboard taps: {date, k, t, v} \u2014 v is the tick time (ms) or false to untick
  const date = body.date, k = Number(body.k), t = Number(body.t);
  if (!validDate(date) || !(k >= 0 && k < 20) || !(t >= 0 && t < 50)) return json({ error: "bad input" });
  const v = typeof body.v === "number" ? body.v : (body.v ? Date.now() : false);
  return json({ checks: setTick(date, k, t, v) });
}

function handleCheck(b) {
  const k = Number(b.k), t = Number(b.t);
  if (!(k >= 0 && k < KIDS.length) || !(t >= 0 && t < TASK_NAMES.length)) return json({ ok: false, message: "Unknown kid or task" });
  const tz = Session.getScriptTimeZone(), now = new Date();
  const day = Number(Utilities.formatDate(now, tz, "u")), hm = Utilities.formatDate(now, tz, "HH:mm");
  if (day > 5 || hm < SCHOOL_WINDOW.start || hm > SCHOOL_WINDOW.end) {
    return json({ ok: false, message: "Not school-run time \u2014 nothing ticked" });
  }
  const date = Utilities.formatDate(now, tz, "yyyy-MM-dd");
  const existing = (readChecks(date)[k] || [])[t];
  if (existing) return json({ ok: true, message: KIDS[k] + " already " + TASK_NAMES[t] + " at " + fmtTime(existing) });
  setTick(date, k, t, now.getTime());
  return json({ ok: true, message: KIDS[k] + " " + TASK_NAMES[t] + " at " + fmtTime(now.getTime()) });
}

function setTick(date, k, t, v) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  let checks, wasDone;
  try {
    checks = readChecks(date);
    checks[k] = checks[k] || [];
    wasDone = !!checks[k][t];
    checks[k][t] = v;
    PropertiesService.getScriptProperties().setProperty("checks-" + date, JSON.stringify(checks));
    cleanup();
  } finally {
    lock.releaseLock();
  }
  if (t === ARRIVED_TASK && v && !wasDone && KIDS[k]) {
    sendAlert("\uD83C\uDFEB " + KIDS[k] + " arrived at school at " + fmtTime(typeof v === "number" ? v : Date.now()));
  }
  return checks;
}

function sendAlert(msg) {
  const results = [];
  if (NTFY_TOPIC) {
    // ntfy can briefly refuse busy shared servers (code 429), so retry a few times before giving up.
    for (let attempt = 1; attempt <= 3; attempt++) {
      let code = 0, body = "";
      try {
        const res = UrlFetchApp.fetch("https://ntfy.sh/" + encodeURIComponent(NTFY_TOPIC),
          { method: "post", payload: msg, muteHttpExceptions: true,
            headers: { Title: "Delgado Family", Tags: "house", Priority: "high" } });
        code = res.getResponseCode(); body = res.getContentText().slice(0, 120);
      } catch (err) { body = String(err).slice(0, 120); }
      results.push("ntfy try " + attempt + ": " + (code || "error") + (code === 200 ? "" : " " + body));
      if (code === 200) break;
      Utilities.sleep(attempt * 2500);
    }
  }
  if (ALERT_EMAIL) {
    try { MailApp.sendEmail(ALERT_EMAIL, msg, msg + "\n\n\u2014 Delgado Family Dashboard"); results.push("email: sent"); }
    catch (err) { results.push("email: " + String(err).slice(0, 120)); }
  }
  // Keep the last 15 alert attempts so problems can be checked (see Script Properties: alert-log).
  try {
    const props = PropertiesService.getScriptProperties();
    const log = JSON.parse(props.getProperty("alert-log") || "[]");
    log.unshift(fmtTime(Date.now()) + " | " + msg.slice(0, 60) + " | " + results.join(" ; "));
    props.setProperty("alert-log", JSON.stringify(log.slice(0, 15)));
  } catch (err) {}
  console.log(results.join(" ; "));
  return results;
}

function fmtTime(ms) { return Utilities.formatDate(new Date(ms), Session.getScriptTimeZone(), "h:mm a"); }

// ---------- Recipe \u2192 ingredients ----------
function handleRecipe(b) {
  let title = "", lines = [];
  if (typeof b.text === "string" && b.text.trim()) {
    title = typeof b.title === "string" && b.title.trim() ? b.title.trim().slice(0, 40) : "Recipe";
    lines = b.text.split(/\r?\n/);
  } else {
    const url = String(b.url || "").trim();
    if (!/^https?:\/\/\S+$/i.test(url) || url.length > 2000) return json({ ok: false, message: "That doesn't look like a web link." });
    let res, why = "";
    // Try like a phone browser first, then plain (some setups reject custom headers).
    const tries = [
      { muteHttpExceptions: true, followRedirects: true, headers: {
          "User-Agent": "Mozilla/5.0 (Linux; Android 14) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/130.0 Mobile Safari/537.36",
          "Accept": "text/html,application/xhtml+xml", "Accept-Language": "en-US,en;q=0.9" } },
      { muteHttpExceptions: true, followRedirects: true }
    ];
    for (let i = 0; i < tries.length && !res; i++) {
      try { res = UrlFetchApp.fetch(url, tries[i]); } catch (err) { why = String(err && err.message || err); console.log("Recipe fetch failed: " + why); }
    }
    if (!res) {
      const perm = /permission|authoriz/i.test(why);
      return json({ ok: false, message: perm
        ? "The Google script isn't allowed to open web pages yet. In the script editor, run testRecipe once and tap Allow. (" + why.slice(0, 120) + ")"
        : "Couldn't open that page (" + why.slice(0, 120) + "). Paste the ingredients instead." });
    }
    if (res.getResponseCode() >= 400) return json({ ok: false, message: "That site wouldn't let us read the page (error " + res.getResponseCode() + "). Paste the ingredients instead." });
    const found = parseRecipe(res.getContentText());
    if (!found || !found.lines.length) return json({ ok: false, message: "Couldn't find an ingredient list on that page. Paste the ingredients instead." });
    title = found.title || "Recipe";
    lines = found.lines;
  }
  lines = lines.map(function (l) { return decodeHtml(stripTags(String(l))).replace(/\s+/g, " ").replace(/^[-\u2022*\u25A2\u25A1\u2610\s]+/, "").trim(); })
               .filter(function (l) { return l && l.length < 200; }).slice(0, 80);
  if (!lines.length) return json({ ok: false, message: "No ingredients found." });

  let items = null, smart = false;
  if (ANTHROPIC_API_KEY) {
    try { items = smartClean(title, lines); smart = true; } catch (err) { console.log("Smart cleanup failed: " + err); }
  }
  if (!items) items = basicClean(lines);
  return json({ ok: true, title: title.slice(0, 40), items: items, smart: smart });
}

function parseRecipe(html) {
  const blocks = html.match(/<script[^>]*type=["']application\/ld\+json["'][^>]*>[\s\S]*?<\/script>/gi) || [];
  for (let i = 0; i < blocks.length; i++) {
    const raw = blocks[i].replace(/^<script[^>]*>/i, "").replace(/<\/script>$/i, "").replace(/<!\[CDATA\[|\]\]>/g, "").trim();
    let data;
    try { data = JSON.parse(raw); } catch (err) { continue; }
    const r = findRecipe(data, 0);
    if (r && Array.isArray(r.recipeIngredient || r.ingredients)) {
      return { title: decodeHtml(String(r.name || "")).trim(), lines: (r.recipeIngredient || r.ingredients).map(String) };
    }
  }
  // Older sites: microdata
  const micro = html.match(/itemprop=["'](?:recipeIngredient|ingredients)["'][^>]*>[\s\S]*?<\/(?:li|span|p|div)>/gi);
  if (micro && micro.length) {
    const t = (html.match(/<title[^>]*>([\s\S]*?)<\/title>/i) || [])[1] || "";
    return { title: decodeHtml(t).split(/[|\-\u2013]/)[0].trim(), lines: micro.map(function (m) { return m.replace(/^[^>]*>/, ""); }) };
  }
  return null;
}

function findRecipe(node, depth) {
  if (!node || depth > 6) return null;
  if (Array.isArray(node)) {
    for (let i = 0; i < node.length; i++) { const r = findRecipe(node[i], depth + 1); if (r) return r; }
    return null;
  }
  if (typeof node !== "object") return null;
  const t = node["@type"];
  if (t === "Recipe" || (Array.isArray(t) && t.indexOf("Recipe") >= 0)) return node;
  if (node["@graph"]) return findRecipe(node["@graph"], depth + 1);
  if (node.mainEntity) return findRecipe(node.mainEntity, depth + 1);
  return null;
}

function stripTags(s) { return s.replace(/<[^>]*>/g, " "); }
function decodeHtml(s) {
  return s.replace(/&#(\d+);/g, function (_, n) { return String.fromCharCode(Number(n)); })
          .replace(/&#x([0-9a-f]+);/gi, function (_, n) { return String.fromCharCode(parseInt(n, 16)); })
          .replace(/&nbsp;/g, " ").replace(/&amp;/g, "&").replace(/&quot;/g, '"').replace(/&#039;|&apos;/g, "'")
          .replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&frac12;/g, "\u00BD").replace(/&frac14;/g, "\u00BC").replace(/&frac34;/g, "\u00BE");
}

/** True for plain salt / pepper / water lines ("salt and pepper to taste", "1 cup warm water"), not "bell pepper". */
function isSkipped(line) {
  const filler = /\b(cups?|c|tbsps?|tablespoons?|tsps?|teaspoons?|pinch(es)?|dash(es)?|to|taste|as|needed|kosher|sea|fine|coarse|flaky|freshly|fresh|ground|black|white|cracked|cold|warm|hot|ice|iced|lukewarm|boiling|room|temperature|and|or|of|for|more|divided|plus|extra|about|optional|tap|filtered|seasoning|some|a|few|grinds?|ml|liters?|l|oz|ounces?)\b/g;
  const words = line.toLowerCase().replace(/\([^)]*\)/g, " ").replace(/[\d\u00BD\u00BC\u00BE\u2153\u2154\u215B\/.,+\-\u2013:;*]/g, " ").replace(filler, " ")
                    .split(/\s+/).filter(Boolean);
  return words.length > 0 && words.every(function (w) { return SKIP_INGREDIENTS.indexOf(w) >= 0 || w === "peppercorns"; });
}

function basicClean(lines) {
  const seen = {};
  return lines.filter(function (l) {
    if (isSkipped(l)) return false;
    const k = l.toLowerCase(); if (seen[k]) return false; seen[k] = true; return true;
  }).map(function (l) { return { text: l.slice(0, 80), section: "" }; });
}

function smartClean(title, lines) {
  const prompt =
    "Turn these recipe ingredient lines into a grocery shopping list.\n" +
    "Rules:\n" +
    "- One entry per thing to buy. Combine duplicates.\n" +
    "- Store-friendly name first, then the amount in parentheses, e.g. \"Diced tomatoes (2 \u00D7 14.5 oz cans)\", \"Yellow onion (1)\", \"All-purpose flour (2 cups)\".\n" +
    "- Drop prep words like chopped, minced, softened, divided, drained.\n" +
    "- Leave out " + SKIP_INGREDIENTS.join(", ") + " (plain table salt/black pepper/tap water). Keep bell peppers, chili peppers, cayenne and other real items.\n" +
    "- Keep each entry under 70 characters.\n" +
    "- section must be exactly one of: " + AISLES.join(", ") + ".\n" +
    "Reply with only a JSON array like [{\"text\":\"...\",\"section\":\"...\"}] and nothing else.\n\n" +
    "Recipe: " + title + "\nIngredient lines:\n" + lines.map(function (l) { return "- " + l; }).join("\n");
  const res = UrlFetchApp.fetch("https://api.anthropic.com/v1/messages", {
    method: "post", contentType: "application/json", muteHttpExceptions: true,
    headers: { "x-api-key": ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01" },
    payload: JSON.stringify({ model: CLEANUP_MODEL, max_tokens: 2000, messages: [{ role: "user", content: prompt }] })
  });
  if (res.getResponseCode() !== 200) throw new Error("API " + res.getResponseCode() + ": " + res.getContentText().slice(0, 200));
  const out = JSON.parse(res.getContentText());
  const text = (out.content || []).filter(function (c) { return c.type === "text"; }).map(function (c) { return c.text; }).join("");
  const m = text.match(/\[[\s\S]*\]/);
  if (!m) throw new Error("No list in reply");
  const arr = JSON.parse(m[0]);
  const items = [];
  arr.forEach(function (x) {
    if (!x || typeof x.text !== "string" || !x.text.trim()) return;
    const t = x.text.trim().slice(0, 80);
    if (isSkipped(t)) return;
    items.push({ text: t, section: AISLES.indexOf(x.section) >= 0 ? x.section : "Other" });
  });
  items.sort(function (a, c) { return AISLES.indexOf(a.section) - AISLES.indexOf(c.section); });
  if (!items.length) throw new Error("Empty list");
  return items;
}

// ---------- Countdowns ----------
function readCountdowns() {
  const raw = PropertiesService.getScriptProperties().getProperty("countdowns");
  const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  return (raw ? JSON.parse(raw) : DEFAULT_COUNTDOWNS).filter(function (c) { return c.date >= today; });
}

function handleCountdownOp(b) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    let list = readCountdowns();
    if (b.op === "cdAdd") {
      const c = b.item || {};
      if (typeof c.id !== "string" || c.id.length > 40 || typeof c.name !== "string" || !c.name.trim() ||
          c.name.length > 40 || !validDate(c.date)) return json({ error: "bad countdown" });
      if (list.length >= 20) return json({ error: "too many countdowns" });
      if (!list.some(function (x) { return x.id === c.id; })) list.push({ id: c.id, name: c.name.trim(), date: c.date });
    } else if (b.op === "cdDel") {
      list = list.filter(function (x) { return x.id !== b.id; });
    } else return json({ error: "bad op" });
    list.sort(function (a, c) { return a.date.localeCompare(c.date); });
    PropertiesService.getScriptProperties().setProperty("countdowns", JSON.stringify(list));
    return json({ ok: true, countdowns: list });
  } finally {
    lock.releaseLock();
  }
}

// ---------- Mom's shifts ----------
function readShifts() {
  const today = Utilities.formatDate(new Date(Date.now() - 864e5), Session.getScriptTimeZone(), "yyyy-MM-dd");
  return JSON.parse(PropertiesService.getScriptProperties().getProperty("shifts") || "[]")
    .filter(function (x) { return x.date >= today; });       // drop old shifts automatically
}

function handleShiftOp(b) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    let list = readShifts();
    if (b.op === "shiftAdd") {
      const x = b.shift || {}, hm = /^\d{2}:\d{2}$/;
      if (typeof x.id !== "string" || x.id.length > 40 || !validDate(x.date) || !hm.test(x.start) || !hm.test(x.end) ||
          WORK_LOCATIONS.indexOf(x.loc) < 0 || typeof x.who !== "string" || x.who.length > 30) return json({ error: "bad shift" });
      if (list.length >= 60) return json({ error: "too many shifts" });
      if (!list.some(function (y) { return y.id === x.id; }))
        list.push({ id: x.id, date: x.date, start: x.start, end: x.end, loc: x.loc, who: x.who, ts: Date.now() });
    } else if (b.op === "shiftDel") {
      list = list.filter(function (y) { return y.id !== b.id; });
    } else return json({ error: "bad op" });
    list.sort(function (a, c) { return (a.date + a.start).localeCompare(c.date + c.start); });
    PropertiesService.getScriptProperties().setProperty("shifts", JSON.stringify(list));
    return json({ ok: true, shifts: list });
  } finally {
    lock.releaseLock();
  }
}

// ---------- Mom's work arrive / leave (sent by her Android phone) ----------
function readWork(date) {
  if (!validDate(date)) return [];
  return JSON.parse(PropertiesService.getScriptProperties().getProperty("work-" + date) || "[]");
}

function handleWork(b) {
  if (["arrive", "leave"].indexOf(b.event) < 0 || WORK_LOCATIONS.indexOf(b.loc) < 0) return json({ ok: false, message: "bad work event" });
  const tz = Session.getScriptTimeZone(), now = Date.now(), date = Utilities.formatDate(new Date(now), tz, "yyyy-MM-dd");
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  let skip = false;
  try {
    const list = readWork(date);
    const last = list[list.length - 1];
    // ignore GPS flapping: same event at the same place within 10 minutes
    if (last && last.event === b.event && last.loc === b.loc && now - last.ts < 10 * 60000) skip = true;
    else {
      list.push({ ts: now, event: b.event, loc: b.loc });
      PropertiesService.getScriptProperties().setProperty("work-" + date, JSON.stringify(list));
      cleanup();
    }
  } finally {
    lock.releaseLock();
  }
  if (skip) return json({ ok: true, message: "duplicate ignored" });
  const shift = readShifts().filter(function (x) { return x.date === date && x.loc === b.loc; })[0];
  const note = shift ? " (shift " + niceTime(shift.start) + "\u2013" + niceTime(shift.end) + ")" : " (no shift on the schedule)";
  const msg = (b.event === "arrive" ? "\uD83C\uDFE5 Mom arrived at " : "\uD83D\uDE97 Mom left ") + b.loc + " at " + fmtTime(now) + note;
  sendAlert(msg);
  return json({ ok: true, message: msg });
}

function niceTime(hm) {
  const p = hm.split(":").map(Number), h = p[0] % 12 || 12;
  return h + ":" + (p[1] < 10 ? "0" : "") + p[1] + (p[0] < 12 ? "a" : "p");
}

function handleLogOp(b) {
  if (!validDate(b.date)) return json({ error: "bad date" });
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const key = "log-" + b.date, props = PropertiesService.getScriptProperties();
    const arr = JSON.parse(props.getProperty(key) || "[]");
    let out = arr;
    if (b.op === "log") {
      const en = b.entry || {};
      if (typeof en.id !== "string" || en.id.length > 40 || LOG_TYPES.indexOf(en.type) < 0 ||
          typeof en.who !== "string" || en.who.length > 30 || typeof en.ts !== "number" ||
          !(en.end === null || typeof en.end === "number")) return json({ error: "bad entry" });
      if (!arr.some(function (x) { return x.id === en.id; })) arr.push({ id: en.id, type: en.type, who: en.who, ts: en.ts, end: en.end });
    } else if (b.op === "unlog") {
      out = arr.filter(function (x) { return x.id !== b.id; });
    } else if (b.op === "update") {
      if (typeof b.end !== "number") return json({ error: "bad end" });
      arr.forEach(function (x) { if (x.id === b.id) x.end = b.end; });
    } else return json({ error: "bad op" });
    props.setProperty(key, JSON.stringify(out));
    cleanup();
    return json({ ok: true });
  } finally {
    lock.releaseLock();
  }
}

function readShop() {
  return JSON.parse(PropertiesService.getScriptProperties().getProperty("shop") || "[]");
}

function handleShopOp(b) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    let list = readShop();
    if (b.op === "shopAdd") {
      const it = b.item || {};
      if (typeof it.id !== "string" || it.id.length > 40 || typeof it.text !== "string" || !it.text.trim() ||
          it.text.length > 80 || typeof it.who !== "string" || it.who.length > 30 || typeof it.ts !== "number") return json({ error: "bad item" });
      if (list.length >= 150) return json({ error: "list full" });
      if (!list.some(function (x) { return x.id === it.id; })) list.push({ id: it.id, text: it.text.trim(), who: it.who, ts: it.ts, done: false });
    } else if (b.op === "shopAddMany") {
      const arr = Array.isArray(b.items) ? b.items.slice(0, 60) : [];
      arr.forEach(function (it) {
        if (!it || typeof it.id !== "string" || it.id.length > 40 || typeof it.text !== "string" || !it.text.trim() ||
            it.text.length > 80 || typeof it.who !== "string" || it.who.length > 30 || typeof it.ts !== "number") return;
        if (list.length >= 150 || list.some(function (x) { return x.id === it.id; })) return;
        list.push({ id: it.id, text: it.text.trim(), who: it.who, ts: it.ts, done: false,
                    from: typeof it.from === "string" ? it.from.slice(0, 40) : "",
                    section: AISLES.indexOf(it.section) >= 0 ? it.section : "" });
      });
    } else if (b.op === "shopToggle") {
      list.forEach(function (x) { if (x.id === b.id) x.done = !!b.done; });
    } else if (b.op === "shopDel") {
      list = list.filter(function (x) { return x.id !== b.id; });
    } else if (b.op === "shopClear") {
      list = list.filter(function (x) { return !x.done; });
    } else return json({ error: "bad op" });
    PropertiesService.getScriptProperties().setProperty("shop", JSON.stringify(list));
    return json({ ok: true });
  } finally {
    lock.releaseLock();
  }
}

function readLogs(date, days) {
  if (!validDate(date)) return {};
  const p = date.split("-").map(Number), props = PropertiesService.getScriptProperties(), out = {};
  for (let i = days - 1; i >= 0; i--) {
    const d = Utilities.formatDate(new Date(Date.UTC(p[0], p[1] - 1, p[2] - i)), "UTC", "yyyy-MM-dd");
    out[d] = JSON.parse(props.getProperty("log-" + d) || "[]");
  }
  return out;
}

function readChecks(date) {
  if (!validDate(date)) return {};
  const raw = PropertiesService.getScriptProperties().getProperty("checks-" + date);
  return raw ? JSON.parse(raw) : {};
}

function getEvents() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get("events");
  if (hit) return JSON.parse(hit);

  const tz = Session.getScriptTimeZone();
  const start = new Date(); start.setHours(0, 0, 0, 0);
  const end = new Date(start.getTime() + DAYS_AHEAD * 864e5);
  const out = [];

  CALENDAR_IDS.forEach(function (id) {
    const cal = id === "primary" ? CalendarApp.getDefaultCalendar() : CalendarApp.getCalendarById(id);
    if (!cal) return;
    cal.getEvents(start, end).forEach(function (ev) {
      if (ev.isAllDayEvent()) {
        out.push({
          title: ev.getTitle(), allDay: true,
          start: Utilities.formatDate(ev.getAllDayStartDate(), tz, "yyyy-MM-dd"),
          end: Utilities.formatDate(ev.getAllDayEndDate(), tz, "yyyy-MM-dd")
        });
      } else {
        out.push({ title: ev.getTitle(), allDay: false, start: ev.getStartTime().getTime(), end: ev.getEndTime().getTime() });
      }
    });
  });

  cache.put("events", JSON.stringify(out), 300);   // 5-minute cache keeps it fast
  return out;
}

function cleanup() {
  const props = PropertiesService.getScriptProperties();
  const cutoff = Utilities.formatDate(new Date(Date.now() - KEEP_DAYS * 864e5), Session.getScriptTimeZone(), "yyyy-MM-dd");
  const logCutoff = Utilities.formatDate(new Date(Date.now() - KEEP_LOG_DAYS * 864e5), Session.getScriptTimeZone(), "yyyy-MM-dd");
  props.getKeys().forEach(function (key) {
    if (key.indexOf("checks-") === 0 && key.slice(7) < cutoff) props.deleteProperty(key);
    if (key.indexOf("log-") === 0 && key.slice(4) < logCutoff) props.deleteProperty(key);
    if (key.indexOf("work-") === 0 && key.slice(5) < logCutoff) props.deleteProperty(key);
  });
}

function validDate(s) { return typeof s === "string" && /^\d{4}-\d{2}-\d{2}$/.test(s); }

function json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
